\begin{abstract}
We present a five-dimensional geometric singular perturbation model for the stable
boundary layer with state variables
\((\tilde{e}, q_\theta, S, T_s, T_g)\).
The model combines desingularized turbulence dynamics, slaved thermodynamic
closures, and slow thermal coupling to support fold-locus diagnostics and
data-driven parameter identification.
\end{abstract}

\section{Closures and Smooth Regularization}
To maintain \(C^r\)-smoothness near neutral stratification and support robust
implicit integration with autodiff Jacobians, the stratification closure uses a
hyperbolic embedding of \(\max(0,x)\):
\begin{equation}
\operatorname{smooth\_max}(x;\epsilon) = \frac{x + \sqrt{x^2 + \epsilon^2}}{2}.
\end{equation}

With
\(x = (\theta_0-T_s)/h_{\text{sbl}}(\tilde{e})\),
the closure becomes
\begin{equation}
\theta_z(T_s,\tilde{e}) = \operatorname{smooth\_max}\!\left(\frac{\theta_0-T_s}{h_{\text{sbl}}(\tilde{e})};\epsilon_{\text{strat}}\right),
\end{equation}
where \(\epsilon_{\text{strat}}\) is a small smoothing scale exposed in
\texttt{SBLParams}.

The bulk transfer coefficients are
\begin{align}
C_D &= \left[\frac{\kappa}{\ln((z_1-d)/z_{0m})}\right]^2, \\
C_H &= \frac{\kappa^2}{\ln((z_1-d)/z_{0m})\,\ln((z_1-d)/z_{0h})}.
\end{align}

\section{Governing Equations}
The desingularized fast-slow system in fast time \(\tau\) is
\begin{align}
\frac{d\tilde{e}}{d\tau} &= \frac{1}{2} c_m \ell S^2 \tilde{e} - \frac{g}{2\theta_0} q_\theta - \frac{1}{2\ell} \tilde{e}^3, \\
\frac{dq_\theta}{d\tau} &= -c_w \theta_z(T_s,\tilde{e})\tilde{e}^3 - \frac{g}{\theta_0}c_\theta\ell q_\theta^2 - \frac{C_\theta}{\ell}\tilde{e}^2 q_\theta, \\
\frac{dS}{d\tau} &= \frac{\epsilon_1}{\max(\tilde{e},\sqrt{\delta})}\,\mathcal{F}_S, \\
\frac{dT_s}{d\tau} &= \frac{\epsilon_1}{\max(\tilde{e},\sqrt{\delta})}\frac{1}{C_s}\left(R_{\text{net}} + \rho c_p q_\theta + \frac{k_g}{d_g}(T_g-T_s)\right), \\
\frac{dT_g}{d\tau} &= \frac{\epsilon_1\epsilon_2}{\max(\tilde{e},\sqrt{\delta})}\frac{\kappa_g}{d_g^2}(T_s-T_g).
\end{align}

The interior mixing length is
\begin{equation}
\ell(z_{\text{eff}})=\frac{\kappa z_{\text{eff}}}{1+\kappa z_{\text{eff}}/\ell_\infty},
\quad z_{\text{eff}} = z-d+z_{0m}.
\end{equation}
